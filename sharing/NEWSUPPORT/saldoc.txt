CREATE TABLE Logs (
    statusNumber INT IDENTITY(1,1) PRIMARY KEY, -- Auto-increment
    issueNumber UNIQUEIDENTIFIER,
    statusId UNIQUEIDENTIFIER,
    RaisedDate DATETIME,
    ResolvedDate DATETIME
);
