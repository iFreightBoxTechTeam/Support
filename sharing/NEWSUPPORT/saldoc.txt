CREATE TABLE Logs (
    statusNumber INT IDENTITY(1,1) PRIMARY KEY, -- Auto-increment
    issueNumber UNIQUEIDENTIFIER,
    statusId UNIQUEIDENTIFIER,
    RaisedDate DATETIME,
    ResolvedDate DATETIME
);

CREATE TRIGGER trg_AfterInsert_Issues
ON issues
AFTER INSERT
AS
BEGIN
    SET NOCOUNT ON;

    INSERT INTO Logs (issueNumber, statusId, RaisedDate, ResolvedDate)
    SELECT 
        i.issues_id,
        i.statusid,
        i.Raised_date,
        i.resolve_date
    FROM inserted i;
END;
